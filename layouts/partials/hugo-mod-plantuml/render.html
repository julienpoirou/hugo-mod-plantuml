{{- /* Shared renderer for the `plantuml` and `puml` shortcodes. Receives the calling shortcode context ("."), resolves the diagram source from one of three inputs, and emits the client-side render wrapper (plus the shared assets, once per page). Inputs (shortcode params): inner content - {{< plantuml >}} @startuml ... @enduml {{< /plantuml >}} src="path.puml" - read from the site's assets/ tree b64="..." - pre-encoded base64 source dark="true" - render in dark mode class="..." - extra CSS class on the wrapper */ -}} {{- $source := or .Inner "" -}} {{- $src := or (.Get "src") "" -}} {{- $encodedSource := "" -}} {{- with .Get "b64" -}} {{- $encodedSource = . -}} {{- end -}} {{- if ne $src "" -}} {{- $source = readFile (printf "assets/%s" $src) -}} {{- end -}} {{- if eq $encodedSource "" -}} {{- $source = trim $source " \n\r\t" -}} {{- if eq $source "" -}} {{- errorf "The %q shortcode requires one of: inner content, b64, or src. File: %s" .Name .Page.File.Path -}} {{- end -}} {{- $encodedSource = $source | base64Encode -}} {{- end -}} {{- $dark := in (slice "true" "1" "yes") (lower (default "" (.Get "dark"))) -}} {{- $class := default "" (.Get "class") -}} {{- $engine := resources.Get "libs/hugo-mod-plantuml/plantuml.js" | fingerprint "sha256" -}} {{- if not (.Page.Scratch.Get "hugoModPlantUMLLoaded") -}} {{- $viz := resources.Get "libs/hugo-mod-plantuml/viz-global.js" | fingerprint "sha256" -}} {{- $style := resources.Get "libs/hugo-mod-plantuml/hugo-mod-plantuml.css" | fingerprint "sha256" -}} {{- $glue := resources.Get "libs/hugo-mod-plantuml/hugo-mod-plantuml.js" | fingerprint "sha256" -}} {{- .Page.Scratch.Set "hugoModPlantUMLLoaded" true -}} {{- end -}}
